一个经典结论：
>对一组点按照 DFS 序 tin 排序，只需要计算相邻两点的 LCA，就能够找出虚树中所有额外的 LCA 节点。

模板一：
```C++
struct VirtualTree {
    static const int LOG = 20;

    int n, timer;
    vector<vector<int>> g, vg;
    vector<int> dep, tin, tout;
    vector<array<int, LOG>> up;

    VirtualTree(int n) : n(n), timer(0), g(n + 1), vg(n + 1),
        dep(n + 1), tin(n + 1), tout(n + 1), up(n + 1) {}

    void add_edge(int x, int y)
    {
        g[x].push_back(y);
        g[y].push_back(x);
    }

    void dfs(int x, int fa)
    {
        tin[x] = ++timer;

        up[x][0] = fa;

        for(int j = 1; j < LOG; j++)
            up[x][j] = up[up[x][j - 1]][j - 1];

        for(int y : g[x]){
            if(y == fa) continue;

            dep[y] = dep[x] + 1;
            dfs(y, x);
        }

        tout[x] = timer;
    }

    void init(int root = 1)
    {
        dep[root] = 0;
        dfs(root, root);
    }

    bool is_ancestor(int x, int y)
    {
        return tin[x] <= tin[y] && tout[y] <= tout[x];
    }

    int lca(int x, int y)
    {
        if(is_ancestor(x, y)) return x;
        if(is_ancestor(y, x)) return y;

        for(int j = LOG - 1; j >= 0; j--){
            if(!is_ancestor(up[x][j], y))
                x = up[x][j];
        }

        return up[x][0];
    }

    vector<int> build(vector<int> nodes)
    {
        if(nodes.empty())
            return {};

        sort(nodes.begin(), nodes.end(), [&](int x, int y){
            return tin[x] < tin[y];
        });

        int m = nodes.size();

        for(int i = 1; i < m; i++)
            nodes.push_back(lca(nodes[i - 1], nodes[i]));

        sort(nodes.begin(), nodes.end(), [&](int x, int y){
            return tin[x] < tin[y];
        });

        nodes.erase(unique(nodes.begin(), nodes.end()), nodes.end());

        for(int x : nodes)
            vg[x].clear();

        vector<int> st;

        for(int x : nodes){
            while(!st.empty() && !is_ancestor(st.back(), x))
                st.pop_back();

            if(!st.empty())
                vg[st.back()].push_back(x);

            st.push_back(x);
        }

        return nodes;
    }
};

//使用：
VirtualTree vt(n);

for(int i = 1; i < n; i++){
    int u, v;
    cin >> u >> v;
    vt.add_edge(u, v);
}

vt.init(1);

vector<int> key = {3, 5, 8, 12};

auto nodes = vt.build(key);

int root = nodes[0];

for(int x : nodes){
    for(int y : vt.vg[x]){
        // 虚树边 x -> y
    }
}
```

模板二：
```C++
#include <bits/stdc++.h>
using namespace std;

const int N = 200005;
const int LOG = 20;

vector<int> g[N];

int dep[N];
int tin[N], tout[N], timer;
int up[N][LOG];

// --------------------------------------------------
// 原树预处理
// --------------------------------------------------

void dfs(int x, int fa)
{
    tin[x] = ++timer;

    up[x][0] = fa;

    for(int j = 1; j < LOG; j++)
        up[x][j] = up[up[x][j - 1]][j - 1];

    for(int y : g[x]){
        if(y == fa) continue;

        dep[y] = dep[x] + 1;
        dfs(y, x);
    }

    tout[x] = timer;
}

bool is_ancestor(int x, int y)
{
    return tin[x] <= tin[y] && tout[y] <= tout[x];
}

int lca(int x, int y)
{
    if(is_ancestor(x, y)) return x;
    if(is_ancestor(y, x)) return y;

    for(int j = LOG - 1; j >= 0; j--){
        if(!is_ancestor(up[x][j], y))
            x = up[x][j];
    }

    return up[x][0];
}

// --------------------------------------------------
// 构造虚树
// --------------------------------------------------

vector<int> vg[N]; // 虚树

vector<int> build_virtual_tree(vector<int> nodes)
{
    if(nodes.empty())
        return {};

    // 1. 按 DFS 序排序
    sort(nodes.begin(), nodes.end(), [&](int x, int y){
        return tin[x] < tin[y];
    });

    // 2. 加入相邻节点的 LCA
    int m = nodes.size();

    for(int i = 1; i < m; i++)
        nodes.push_back(lca(nodes[i - 1], nodes[i]));

    // 3. 再次排序 + 去重
    sort(nodes.begin(), nodes.end(), [&](int x, int y){
        return tin[x] < tin[y];
    });

    nodes.erase(unique(nodes.begin(), nodes.end()), nodes.end());

    // 清空这些节点上一轮的虚树边
    for(int x : nodes)
        vg[x].clear();

    // 4. 用栈建树
    vector<int> st;

    for(int x : nodes){
        while(!st.empty() && !is_ancestor(st.back(), x))
            st.pop_back();

        if(!st.empty()){
            // st.back() 是 x 在虚树上的父亲
            vg[st.back()].push_back(x);
        }

        st.push_back(x);
    }

    return nodes;
}
```
